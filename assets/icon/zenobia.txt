# Icon of zenobia.exe, the Windows player: a tarot card in a gold frame with
# a crown and an upright sword on a night-blue face. Original pixel art made
# for this project (MIT licence, see LICENSE); no game graphics, no logos.
#
# tools/release/mkicon.py turns this file into build/zenobia.ico while
# zenobia.exe is built (CMakeLists.txt). The .ico is a build output and is
# never committed. Preview: python tools/release/mkicon.py
# assets/icon/zenobia.txt OUT.ico --preview OUT.png
#
# Format (the text-art images of docs/workstreams/gfx-pipeline.md 1.4, with
# colours given directly instead of hardware palettes):
#   @chars C = COLOUR ...  one character per pixel colour: #RRGGBB (opaque),
#                          #RRGGBBAA, or none (transparent)
#   @size W H              starts an image; @data follows, then H rows of W
#                          characters
#   @scale FROM to TO      also store the FROM x FROM image scaled up to
#                          TO x TO pixels (whole multiples, no smoothing), so
#                          that high-DPI screens show sharp pixels
# Lines starting with # are comments.

@chars  K = #1B1526     # outline
@chars  Y = #FFE48C     # gold, highlight

@chars  N = #2C3C82     # card face, night blue
@chars  n = #202C66     # card face, inner shadow
@chars  W = #FFFFFF     # white: blade highlight, stars
@chars  S = #D2DCEA     # steel, light
@chars  s = #8A98B0     # steel, shade
@chars  R = #E0384C     # ruby
@chars  r = #8C1E30     # ruby, dark; sword grip
